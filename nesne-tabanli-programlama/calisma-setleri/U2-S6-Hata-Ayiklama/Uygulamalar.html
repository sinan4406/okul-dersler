private void btnEkle_Click(object sender, EventArgs e)
{
    int toplam = 0;
    int adet = 0;
    int enYuksek = 0;
    try
    {
        adet++;
        int puan = Convert.ToInt32(txtPuan.Text);
        toplam += puan;
        if (puan > enYuksek)
        {
            enYuksek = puan;
        }
        lstPuan.Items.Add(puan.ToString());
        lblToplam.Text = "Toplam: " + toplam.ToString();
        lblAdet.Text = "Adet: " + adet.ToString();
        lblOrt.Text = "Ortalama: " + (toplam / adet).ToString();
        lblEn.Text = "En yüksek: " + enYuksek.ToString();
    }
    catch
    {
        lblUyari.Text = "Puanı rakamla yaz.";
    }
}

Görev: (1) Sırayla 80, 90, abc, 120, 70 gir; her adımda ne görünüyor, ne görünmesi gerekir? Kesme noktası ve F10 ile adet ve toplam değerlerini izle. (2) Üç hatayı bul, düzelt ve her hatanın neden yanlış olduğunu bir cümleyle yorum satırına yaz. Beklenen: aralık dışı puan (örnekte 120) ve harf listeye eklenmez, sayaçlar değişmez; sonunda Toplam 240, Adet 3, Ortalama 80, En yüksek 90. (3) Kendi test tablonu yaz: en az iki sınır durumu (ör. 0, 100, 101, −1, boş kutu) için girdi ve beklenen sonucu belirt, programla dene.

Ödemeyi kapat: Öğrenci kartı işaretliyse toplamdan %10 (tam sayı bölmesiyle) düşer. Alınan para yetmiyorsa Eksik: … TL, yetiyorsa Para üstü: … TL yazılır ve ödeme kapanınca sepet (liste, kalem sayısı, toplam) sıfırlanır; para yetmiyorsa sepet olduğu gibi kalır.
